// thread_count = 64

struct Camera {
	mat4 viewProjectionMatrix;
};

struct GeometryView {
	uint indexCount;
    uint firstIndex;
    int vertexOffset;
    uint firstInstance;
};

struct RenderGroupInstance {
	uint cameraIndex;
	uint motionType;
	uint startInstanceOffset;
	uint instanceCount; // reset to zero before culling pass
	GeometryView sharedGeometry;
};

struct Instance {
	mat4 worldMatrix;
};

struct IndirectDrawCommand {
	uint indexCount;
    uint instanceCount;
    uint firstIndex;
    int vertexOffset;
    uint firstInstance;
};

readonly buffer Cameras {
	Camera cameras[];
};

readonly buffer RenderGroupInstancesToCull {
	uint indices[];
} renderGroupInstancesToCull;

readonly buffer RenderGroupInstances {
	RenderGroupInstance instances[];
} renderGroupInstances;

readonly buffer StaticRenderInstances {
	Instance instances[];
} staticRenderInstances;

writeonly buffer VisibleStaticRenderInstances {
	uint indices[];
} visibleStaticRenderInstances;

buffer IndirectDrawCount {
	uint count;
} indirectDrawCount;

writeonly buffer IndirectDrawCommands {
	IndirectDrawCommand cmds[];
} indirectDrawCommands;

shared uint groupInstanceCountShared;

void main()
{
	if (threadID == uvec3(0))
		groupInstanceCountShared = 0;
		
	barrier();
	
	uint renderGroupLookup = WorkgroupIDToLinearIndex(workgroupID);
	uint instanceLookup = ThreadIDToLinearIndex(threadID);
	
	uint renderGroupInstanceLookup = renderGroupInstancesToCull.indices[ renderGroupLookup ];
	RenderGroupInstance renderGroupInstance = renderGroupInstances.instances[ renderGroupInstanceLookup ];
	
	Instance instance = staticRenderInstances.instances[ renderGroupInstance.startInstanceOffset + instanceLookup ];
	
	Camera camera = cameras[ renderGroupInstance.cameraIndex ];
	
	if (!IsCulled(instance, camera))
	{
		uint visibleInstanceLookup = atomicAdd(groupInstanceCountShared, 1);
		
		visibleStaticRenderInstances.indices[ renderGroupInstance.startInstanceOffset + visibleInstanceLookup ] =
			renderGroupInstance.startInstanceOffset + instanceLookup;
	}
	
	barrier();
	
	uint groupInstanceVisibleInstanceCount = groupInstanceCountShared;
	
	if (threadID == uvec3(0) &&
		groupInstanceVisibleInstanceCount > 0)
	{
		uint indirectDrawCommandLookup = atomicAdd(indirectDrawCount.count, 1);
		
		IndirectDrawCommand indirectDrawCommand;
		indirectDrawCommand.indexCount = renderGroupInstance.sharedGeometry.indexCount;
		indirectDrawCommand.instanceCount = groupInstanceVisibleInstanceCount;
		indirectDrawCommand.firstIndex = renderGroupInstance.sharedGeometry.firstIndex;
		indirectDrawCommand.vertexOffset = renderGroupInstance.sharedGeometry.vertexOffset;
		indirectDrawCommand.firstInstance = renderGroupInstance.sharedGeometry.firstInstance;
		
		indirectDrawCommands.cmds[ indirectDrawCommandLookup ] = indirectDrawCommand;
	}
}
